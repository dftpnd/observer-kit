# Файл принадлежит T000 и больше не правится: новый тест = новый .cpp рядом со smoke.cpp.
file(GLOB _ovr_unit_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
foreach(_src IN LISTS _ovr_unit_sources)
  get_filename_component(_case "${_src}" NAME_WE)
  add_executable("ovr_${_case}" "${_src}")
  target_link_libraries("ovr_${_case}" PRIVATE ovr_lib)
  if(NOT _case STREQUAL "test_golden_state")
    add_test(NAME "${_case}" COMMAND "ovr_${_case}" WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
  endif()
endforeach()

file(GLOB _ovr_python_tests CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/test_*.py")
foreach(_src IN LISTS _ovr_python_tests)
  get_filename_component(_case "${_src}" NAME_WE)
  set(_args)
  if(_case STREQUAL "test_golden_state")
    list(APPEND _args "$<TARGET_FILE:ovr_test_golden_state>")
  endif()
  add_test(NAME "${_case}" COMMAND "${Python3_EXECUTABLE}" -B "${_src}" ${_args}
           WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
endforeach()
